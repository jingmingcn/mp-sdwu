## 第一节 wx.request 的原始形态

---

## 1.1 原始 API

```js
const task = wx.request({
  url: 'https://api.example.com/goods',
  method: 'GET',
  data: { page: 1 },
  header: { 'content-type': 'application/json' },
  timeout: 10000,
  success(res) { console.log(res.statusCode, res.data, res.header) },
  fail(err) { console.error(err.errMsg) },
  complete() {}
})

task.abort()                        // 取消
task.onHeadersReceived(cb)          // 收到响应头
```

---

## 1.2 五个必须知道的约束

<!-- _class: smalltext -->

| 约束 | 数值 | 影响 |
|---|---|---|
| 并发上限 | **10** 个（含 uploadFile/downloadFile） | 超出会排队，长列表图片预加载易打满 |
| 超时默认 | 60s（可在 `app.json` 的 `networkTimeout` 改） | 默认太长，用户会以为卡死 |
| 域名白名单 | 必须在 MP 后台配置 **request 合法域名** | 且必须 HTTPS、备案 |
| 单次数据量 | 上行/下行都有限制（约 10MB） | 大文件必须走 upload/download |
| 无 Cookie | 不自动带 Cookie | 登录态必须自己放 header |

<br>

**开发期**可在开发者工具勾选"不校验合法域名"，
但**真机预览与体验版一样会校验**——很多人到提测才发现域名没配。

---

## 1.3 fail 不等于业务失败

```js
wx.request({
  url, 
  success(res) {
    // ⚠ HTTP 500、404 也会走 success！
    if (res.statusCode !== 200) { /* 需要自己判断 */ }
  },
  fail(err) {
    // 只有网络层失败才走这里：
    // 域名不合法 / 超时 / DNS 失败 / 断网 / 证书错误 / abort
  }
})
```

<br>

**三层错误要分清**：

1. **网络层**（`fail`）：断网、超时、域名非法 → 提示"网络异常，请重试"
2. **HTTP 层**（`statusCode`）：4xx/5xx → 401 跳登录，5xx 提示"服务开小差"
3. **业务层**（`res.data.code`）：库存不足、余额不够 → 展示后端给的 message

## 2.1 最小 Promise 封装

```js
// request/core.js
export function rawRequest(options) {
  return new Promise((resolve, reject) => {
    const task = wx.request({
      ...options,
      success: res => resolve(res),
      fail: err => reject(Object.assign(new Error(err.errMsg), { type: 'NETWORK', raw: err }))
    })
    // 把 task 挂到 promise 上，便于外部 abort
    options.onTask && options.onTask(task)
  })
}
```

## 2.2 拦截器的设计

<!-- _class: smalltext -->

```js
// request/interceptor.js
export class InterceptorManager {
  constructor() { this.handlers = [] }
  use(onFulfilled, onRejected) {
    this.handlers.push({ onFulfilled, onRejected })
    return this.handlers.length - 1
  }
  eject(id) { this.handlers[id] = null }
  forEach(fn) { this.handlers.forEach(h => h && fn(h)) }
}
```

```js
// 组装成 Promise 链（与 axios 同构）
function dispatch(config) {
  const chain = [adapter, undefined]
  http.interceptors.request.forEach(h => chain.unshift(h.onFulfilled, h.onRejected))
  http.interceptors.response.forEach(h => chain.push(h.onFulfilled, h.onRejected))

  let promise = Promise.resolve(config)
  while (chain.length) promise = promise.then(chain.shift(), chain.shift())
  return promise
}
```

## 2.3 完整的 http 实例

<!-- _class: tinytext -->

```js
// request/index.js
import { InterceptorManager } from './interceptor'
import { BASE_URL, TIMEOUT } from '../config'

const http = {
  interceptors: {
    request: new InterceptorManager(),
    response: new InterceptorManager()
  },
  request(config) {
    config = { method: 'GET', timeout: TIMEOUT, header: {}, ...config }
    if (!/^https?:/.test(config.url)) config.url = BASE_URL + config.url
    return dispatch(config)
  },
  get(url, data, cfg)  { return http.request({ ...cfg, url, data, method: 'GET' }) },
  post(url, data, cfg) { return http.request({ ...cfg, url, data, method: 'POST' }) },
  put(url, data, cfg)  { return http.request({ ...cfg, url, data, method: 'PUT' }) },
  del(url, data, cfg)  { return http.request({ ...cfg, url, data, method: 'DELETE' }) }
}

function adapter(config) {
  return new Promise((resolve, reject) => {
    const task = wx.request({
      ...config,
      success: res => resolve({ ...res, config }),
      fail: err => reject(Object.assign(new Error(err.errMsg), { type: 'NETWORK', config }))
    })
    config.signal && config.signal.bind(task)
  })
}

export default http
```

## 2.5 响应拦截：三层错误归一

<!-- _class: smalltext -->

```js
http.interceptors.response.use(res => {
  const { statusCode, data } = res

  // HTTP 层
  if (statusCode === 401) {
    return handle401(res.config)                    // 见 2.6
  }
  if (statusCode < 200 || statusCode >= 300) {
    throw Object.assign(new Error(`服务异常(${statusCode})`), { type: 'HTTP', statusCode })
  }

  // 业务层：约定 { code: 0, data, message }
  if (data.code !== 0) {
    throw Object.assign(new Error(data.message || '操作失败'), {
      type: 'BIZ', code: data.code, data: data.data
    })
  }

  return data.data          // ✅ 业务代码直接拿到 data，不用层层解包
})
```

// 业务层用起来非常干净

## 2.6 401 与 token 刷新（防并发风暴）

<!-- _class: smalltext -->

```js
let refreshing = null              // 正在进行的刷新 Promise
const waiters = []                 // 等待刷新完成的请求

async function handle401(config) {
  if (config._retried) { toLogin(); throw new Error('登录已失效') }

  if (!refreshing) {
    refreshing = doRefresh().finally(() => { refreshing = null })
  }
  try {
    await refreshing                       // 多个 401 只触发一次刷新
    config._retried = true
    return http.request(config)            // 用新 token 重放
  } catch (e) {
    toLogin(); throw e
  }
}

async function doRefresh() {
  const { code } = await wxLogin()                     // wx.login
  const { token } = await http.post('/auth/login', { code }, { _skipAuth: true })
  wx.setStorageSync('token', token)
}
```

> 不做并发控制的话，首屏 6 个请求同时 401 会触发 6 次登录 —— 后端会限流封你。

## 第三节 错误处理与重试

---

## 3.1 统一的错误对象

```js
// request/error.js
export class ApiError extends Error {
  constructor({ type, message, code, statusCode, config }) {
    super(message)
    this.name = 'ApiError'
    this.type = type            // 'NETWORK' | 'TIMEOUT' | 'HTTP' | 'BIZ' | 'ABORT'
    this.code = code
    this.statusCode = statusCode
    this.url = config?.url
  }
  get isRetryable() {
    return this.type === 'NETWORK' || this.type === 'TIMEOUT' ||
           (this.type === 'HTTP' && this.statusCode >= 500)
  }
  get userMessage() {
    if (this.type === 'NETWORK') return '网络不给力，请检查后重试'
    if (this.type === 'TIMEOUT') return '请求超时，请重试'
    if (this.type === 'HTTP') return '服务暂时不可用'
    return this.message                       // 业务错误用后端文案
  }
}
```

---

## 3.2 指数退避重试

<!-- _class: smalltext -->

$$
\text{delay}_n = \min(\text{base} \times 2^{n},\ \text{maxDelay}) + \text{jitter}
$$

```js
// request/retry.js
export async function withRetry(fn, { retries = 2, base = 300, maxDelay = 3000 } = {}) {
  let lastErr
  for (let i = 0; i <= retries; i++) {
    try { return await fn() }
    catch (e) {
      lastErr = e
      if (!e.isRetryable || i === retries) throw e
      const jitter = Math.random() * 100                 // 抖动，避免惊群
      const delay = Math.min(base * 2 ** i, maxDelay) + jitter
      await new Promise(r => setTimeout(r, delay))
    }
  }
  throw lastErr
}
```

**只重试幂等请求**：GET 可以重试，POST 下单**绝对不能**——
除非后端支持幂等键（`Idempotency-Key`）。

---

## 3.3 挂到拦截器上

```js
http.interceptors.response.use(null, async err => {
  const { config } = err
  if (!config || config.retry === false) throw err
  if (!err.isRetryable) throw err

  config._retryCount = (config._retryCount || 0) + 1
  if (config._retryCount > (config.retry ?? 2)) throw err

  const delay = Math.min(300 * 2 ** (config._retryCount - 1), 3000) + Math.random() * 100
  await new Promise(r => setTimeout(r, delay))
  return http.request(config)
})
```

```js
await http.get('/goods')                        // 默认重试 2 次
await http.post('/order', body, { retry: false })  // 下单不重试
```

## 3.4 全局兜底

```js
// app.js
App({
  onUnhandledRejection({ reason }) {
    // 未被 catch 的 Promise 错误统一上报
    report('unhandledRejection', reason)
    if (reason?.type === 'NETWORK') {
      wx.showToast({ title: '网络不给力', icon: 'none' })
    }
  },
  onError(stack) { report('jsError', stack) }
})
```

<br>

**原则**：

- 业务代码**只处理它能处理的错误**（如"库存不足"要引导用户换商品）；
- 处理不了的一律 `throw`，交给统一兜底；
- **不要**满屏 `try { } catch (e) { console.log(e) }` —— 那等于吞掉错误。

## 第四节 并发控制与请求去重

---

## 4.1 为什么要限流

小程序的 `wx.request` 并发上限是 **10**。
超出后请求进入队列，**表现为随机的、无规律的慢**。

<br>

典型打满场景：

- 首屏同时请求 12 个模块的数据；
- 长列表滚动时并发预取图片信息；
- 埋点上报没有合并，每个事件一个请求。

---

## 4.2 并发队列

<!-- _class: smalltext -->

```js
// request/queue.js
export class TaskQueue {
  constructor(limit = 6) { this.limit = limit; this.running = 0; this.queue = [] }

  push(taskFn) {
    return new Promise((resolve, reject) => {
      this.queue.push({ taskFn, resolve, reject })
      this._next()
    })
  }

  _next() {
    if (this.running >= this.limit || !this.queue.length) return
    const { taskFn, resolve, reject } = this.queue.shift()
    this.running++
    taskFn()
      .then(resolve, reject)
      .finally(() => { this.running--; this._next() })
  }
}
```

```js
const queue = new TaskQueue(6)     // 留 4 个额度给 upload / 图片
http.interceptors.request.use(config => ({ ...config, _queue: queue }))
```

---

## 4.3 请求去重

<!-- _class: smalltext -->

同一时刻的相同请求（同 url + method + params）应该复用同一个 Promise。

```js
// request/dedupe.js
const inflight = new Map()

function keyOf(config) {
  return [config.method, config.url, JSON.stringify(config.data || {})].join('|')
}

export function dedupe(config, exec) {
  if (config.dedupe === false || config.method !== 'GET') return exec()
  const key = keyOf(config)
  if (inflight.has(key)) return inflight.get(key)

  const p = exec().finally(() => inflight.delete(key))
  inflight.set(key, p)
  return p
}
```

<br>

**典型收益**：页面 A、组件 B、组件 C 都要用户信息，
不做去重就是 3 个请求；去重后 1 个。

---

## 4.4 取消：AbortSignal 风格

<!-- _class: smalltext -->

```js
// request/signal.js
export class RequestSignal {
  constructor() { this.tasks = new Set(); this.aborted = false }
  bind(task) {
    if (this.aborted) return task.abort()
    this.tasks.add(task)
  }
  abort() {
    this.aborted = true
    this.tasks.forEach(t => t.abort())
    this.tasks.clear()
  }
}
```

```js
Page({
  onLoad() {
    this.signal = new RequestSignal()
    http.get('/list', {}, { signal: this.signal })
  },
  onUnload() { this.signal.abort() }      // 页面销毁，取消所有在途请求
})
```

**搜索场景**：每次输入变化先 `abort()` 上一次，再发新请求——
避免"先发的慢请求后到，覆盖了新结果"。

---

## 4.5 竞态：请求序号法

```js
Page({
  data: { list: [] },
  _seq: 0,
  async search(kw) {
    const seq = ++this._seq
    const list = await http.get('/search', { kw })
    if (seq !== this._seq) return          // 已有更新的请求，丢弃本次结果
    this.setData({ list })
  }
})
```

<br>

序号法比 `abort` 更轻，且能处理"请求已完成但结果过期"的情况。
**两者通常一起用**：`abort` 省流量，序号法保正确性。

## 5.1 文件上传

<!-- _class: smalltext -->

```js
export function upload({ url, filePath, name = 'file', formData = {}, onProgress }) {
  return new Promise((resolve, reject) => {
    const task = wx.uploadFile({
      url: BASE_URL + url, filePath, name, formData,
      header: { Authorization: 'Bearer ' + wx.getStorageSync('token') },
      success: res => {
        // ⚠ uploadFile 的 res.data 是「字符串」，必须手动解析
        try {
          const body = JSON.parse(res.data)
          body.code === 0 ? resolve(body.data) : reject(new ApiError(body))
        } catch (e) { reject(new Error('响应解析失败')) }
      },
      fail: err => reject(err)
    })
    task.onProgressUpdate(({ progress }) => onProgress && onProgress(progress))
  })
}
```

```js
const { tempFilePaths } = await wx.chooseMedia({ count: 1, mediaType: ['image'] })
const url = await upload({ url: '/upload', filePath: tempFilePaths[0],
  onProgress: p => this.setData({ percent: p }) })
```

## 5.3 断点续传的思路

小程序**没有内置断点续传**。常规做法是分片：

```js
async function uploadLarge(filePath, chunkSize = 1024 * 1024) {
  const { size } = await wx.getFileInfo({ filePath })
  const total = Math.ceil(size / chunkSize)
  const uploadId = await http.post('/upload/init', { size, total })

  const done = wx.getStorageSync(`upload:${uploadId}`) || []   // 已上传分片
  for (let i = 0; i < total; i++) {
    if (done.includes(i)) continue
    const chunk = await readChunk(filePath, i * chunkSize, chunkSize)  // FileSystemManager
    await upload({ url: '/upload/chunk', filePath: chunk, formData: { uploadId, index: i } })
    done.push(i)
    wx.setStorageSync(`upload:${uploadId}`, done)
  }
  return http.post('/upload/complete', { uploadId })
}
```

## 5.4 WebSocket

<!-- _class: smalltext -->

```js
// request/socket.js
export class Socket {
  constructor(url) { this.url = url; this.handlers = new Map(); this.retry = 0 }

  connect() {
    this.task = wx.connectSocket({ url: this.url })
    this.task.onOpen(() => { this.retry = 0; this._heartbeat() })
    this.task.onMessage(({ data }) => {
      const msg = JSON.parse(data)
      ;(this.handlers.get(msg.type) || []).forEach(fn => fn(msg.payload))
    })
    this.task.onClose(() => this._reconnect())
    this.task.onError(() => this._reconnect())
  }

  _heartbeat() {
    clearInterval(this._hb)
    this._hb = setInterval(() => this.send('ping'), 25000)   // 微信 30s 无数据会断
  }

  _reconnect() {
    clearInterval(this._hb)
    if (this.retry >= 5) return
    setTimeout(() => this.connect(), Math.min(1000 * 2 ** this.retry++, 16000))
  }

  send(type, payload) { this.task?.send({ data: JSON.stringify({ type, payload }) }) }
  on(type, fn) { this.handlers.set(type, [...(this.handlers.get(type) || []), fn]) }
  close() { clearInterval(this._hb); this.retry = 99; this.task?.close() }
}
```

---

## 5.5 WebSocket 的三个要点

1. **心跳必须有**：微信客户端在无数据传输约 30s 后会主动断开，
   服务端也常有空闲超时。心跳间隔取 20–25s。
2. **切后台会断**：`onHide` 时主动 `close()`，`onShow` 时重连。
   否则用户切回来看到的是"连着但收不到消息"。
3. **并发上限**：同时最多 2 个 WebSocket 连接（旧版本为 1 个）。

```js
App({
  onShow() { socket.connect() },
  onHide() { socket.close() }
})
```

## 第六节 本地存储与缓存策略

---

## 6.1 Storage 的能力与限制

| 项 | 值 |
|---|---|
| 单个 key 上限 | 1MB |
| 单个小程序总上限 | **10MB** |
| 作用域 | 按小程序 + 用户维度隔离 |
| 清除时机 | 用户主动清理、长期不用被回收、卸载微信 |
| 同步 API | `wx.setStorageSync` 等，**会阻塞逻辑层** |

<br>

**规则**：

- 启动路径上**只用异步** API；
- 超过 100KB 的数据不要存 Storage，用 `FileSystemManager`；
- **绝不能**把 Storage 当数据库——它随时可能被清空。

---

## 6.2 封装：带过期时间的 Storage

<!-- _class: smalltext -->

```js
// utils/storage.js
const PREFIX = 'app:'

export function setItem(key, value, ttlMs) {
  const record = { v: value, e: ttlMs ? Date.now() + ttlMs : 0 }
  return wx.setStorage({ key: PREFIX + key, data: record })
}

export function getItem(key, defaultValue = null) {
  try {
    const r = wx.getStorageSync(PREFIX + key)
    if (!r || typeof r !== 'object') return defaultValue
    if (r.e && Date.now() > r.e) { wx.removeStorage({ key: PREFIX + key }); return defaultValue }
    return r.v
  } catch (e) { return defaultValue }
}

export function removeItem(key) { return wx.removeStorage({ key: PREFIX + key }) }

// 容量兜底：写入失败（超限）时清理过期项后重试一次
export async function safeSet(key, value, ttl) {
  try { await setItem(key, value, ttl) }
  catch (e) { purgeExpired(); await setItem(key, value, ttl) }
}
```

## 6.3 三种缓存策略

<!-- _class: smalltext -->

| 策略 | 行为 | 适用 |
|---|---|---|
| **Cache First** | 有缓存就用，无则请求 | 静态配置、城市列表、分类树 |
| **Network First** | 先请求，失败降级到缓存 | 订单列表、消息 |
| **Stale While Revalidate** | 立即返回缓存**并**后台刷新 | 首页 feed、商品详情 |

```js
export async function swr(key, fetcher, { ttl = 300_000, onUpdate } = {}) {
  const cached = getItem(key)
  const revalidate = fetcher().then(fresh => {
    safeSet(key, fresh, ttl)
    if (cached && onUpdate) onUpdate(fresh)     // 通知界面二次更新
    return fresh
  })
  return cached ?? revalidate                    // 有缓存就秒开
}
```

## 6.4 SWR 在页面里的用法

```js
Page({
  async onLoad() {
    const data = await swr('home:feed', () => http.get('/feed'), {
      ttl: 5 * 60_000,
      onUpdate: fresh => this.setData({ list: fresh, fromCache: false })
    })
    this.setData({ list: data, fromCache: true })
  }
})
```

<br>

**效果**：二次进入页面 **0ms 出内容**，
数据回来后静默替换。这是首屏体验最划算的一笔投入。

**注意**：更新时不要让列表跳动——保持滚动位置，或只在数据确实变了时才 setData。

## 本章小结

<!-- _class: smalltext -->

1. `wx.request` 的 `success` 不代表成功，要分**网络 / HTTP / 业务**三层处理错误；
2. 封装的核心是**拦截器链**：注入 token、统一 loading、错误归一、401 刷新去重；
3. 重试要用**指数退避 + 抖动**，且**只重试幂等请求**；
4. 并发上限 10：用队列限流、GET 去重、`signal` 取消、序号法防竞态；
5. 上传要**先压缩**，`uploadFile` 的 `res.data` 是字符串；WebSocket 必须有心跳与前后台处理；
6. Storage 上限 10MB 且可能被清空，封装要带 **TTL + 版本号**；
   首屏优先 **SWR**，缓存策略要集中登记。
